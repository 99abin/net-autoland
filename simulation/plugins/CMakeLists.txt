cmake_minimum_required(VERSION 3.16)
project(autoland_launch_assist LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
add_library(AutolandLaunchAssist SHARED LaunchAssist.cc)
target_link_libraries(AutolandLaunchAssist PRIVATE
  gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13)
target_compile_options(AutolandLaunchAssist PRIVATE -Wall -Wextra -Wpedantic)
